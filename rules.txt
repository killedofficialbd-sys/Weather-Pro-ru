rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isSelf(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    // Email stored in Firestore must match the authenticated Firebase Auth token.
    // For providers without an email claim, only an empty email is accepted.
    function authEmailMatches(value) {
      return (('email' in request.auth.token)
          && request.auth.token.email is string
          && value == request.auth.token.email)
        || (!('email' in request.auth.token) && value == '');
    }

    function canonicalChatId(a, b) {
      return a < b ? a + '__' + b : b + '__' + a;
    }

    function acceptedRequestBetween(a, b) {
      return (
        existsAfter(/databases/$(database)/documents/friendRequests/$(a + '_' + b))
        && getAfter(/databases/$(database)/documents/friendRequests/$(a + '_' + b)).data.from == b
        && getAfter(/databases/$(database)/documents/friendRequests/$(a + '_' + b)).data.to == a
        && getAfter(/databases/$(database)/documents/friendRequests/$(a + '_' + b)).data.status == 'accepted'
      ) || (
        existsAfter(/databases/$(database)/documents/friendRequests/$(b + '_' + a))
        && getAfter(/databases/$(database)/documents/friendRequests/$(b + '_' + a)).data.from == a
        && getAfter(/databases/$(database)/documents/friendRequests/$(b + '_' + a)).data.to == b
        && getAfter(/databases/$(database)/documents/friendRequests/$(b + '_' + a)).data.status == 'accepted'
      );
    }

    function isCurrentFriendPair(a, b) {
      return exists(/databases/$(database)/documents/friendships/$(canonicalChatId(a, b)))
        && get(/databases/$(database)/documents/friendships/$(canonicalChatId(a, b))).data.members is list
        && get(/databases/$(database)/documents/friendships/$(canonicalChatId(a, b))).data.members.hasAll([a, b]);
    }

    function chatMember(chatId) {
      return signedIn()
        && exists(/databases/$(database)/documents/chats/$(chatId))
        && get(/databases/$(database)/documents/chats/$(chatId)).data.members is list
        && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.members
        && get(/databases/$(database)/documents/chats/$(chatId)).data.members.size() == 2
        && exists(/databases/$(database)/documents/friendships/$(chatId))
        && get(/databases/$(database)/documents/friendships/$(chatId)).data.members.hasAll(
          get(/databases/$(database)/documents/chats/$(chatId)).data.members
        );
    }

    // Private account data: owner only. A client cannot forge the stored email.
    match /users/{userId} {
      allow read, delete: if isSelf(userId);
      allow create, update: if isSelf(userId)
        && request.resource.data.profile is map
        && authEmailMatches(request.resource.data.profile.email);
    }

    // Searchable public card. Email must always match Firebase Authentication.
    match /publicProfiles/{userId} {
      allow read: if signedIn();
      allow create, update: if isSelf(userId)
        && authEmailMatches(request.resource.data.email);
      allow delete: if isSelf(userId);
    }

    // Request ID convention is recipientUid + "_" + senderUid.
    match /friendRequests/{requestId} {
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly([
          'from', 'to', 'fromName', 'fromAvatar', 'toName', 'status', 'createdAt'
        ])
        && request.resource.data.from == request.auth.uid
        && request.resource.data.to is string
        && request.resource.data.to.size() > 0
        && request.resource.data.to != request.auth.uid
        && requestId == request.resource.data.to + '_' + request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.fromName is string
        && request.resource.data.fromName.size() <= 80
        && request.resource.data.toName is string
        && request.resource.data.toName.size() <= 80
        && request.resource.data.createdAt is timestamp
        && !exists(/databases/$(database)/documents/friendships/$(canonicalChatId(request.auth.uid, request.resource.data.to)))
        && (
          !exists(/databases/$(database)/documents/friendRequests/$(request.auth.uid + '_' + request.resource.data.to))
          || get(/databases/$(database)/documents/friendRequests/$(request.auth.uid + '_' + request.resource.data.to)).data.status in ['declined', 'accepted']
        );

      allow read: if signedIn()
        && (resource.data.from == request.auth.uid || resource.data.to == request.auth.uid);

      // Recipient accepts/declines. Sender may reopen a previously declined request.
      allow update: if signedIn() && (
        (
          resource.data.to == request.auth.uid
          && resource.data.status == 'pending'
          && request.resource.data.from == resource.data.from
          && request.resource.data.to == resource.data.to
          && request.resource.data.status in ['accepted', 'declined']
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'respondedAt'])
          && request.resource.data.respondedAt is timestamp
        ) || (
          resource.data.from == request.auth.uid
          && resource.data.status in ['declined', 'accepted']
          && !exists(/databases/$(database)/documents/friendships/$(canonicalChatId(resource.data.from, resource.data.to)))
          && request.resource.data.from == resource.data.from
          && request.resource.data.to == resource.data.to
          && request.resource.data.status == 'pending'
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'createdAt', 'respondedAt'])
          && request.resource.data.createdAt is timestamp
        )
      );

      // Only sender may cancel a still-pending request.
      allow delete: if signedIn()
        && resource.data.from == request.auth.uid
        && resource.data.status == 'pending';
    }

    // Exactly one canonical friendship document per sorted UID pair.
    match /friendships/{friendshipId} {
      allow read: if signedIn()
        && request.auth.uid in resource.data.members;

      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['members', 'createdAt'])
        && request.resource.data.members is list
        && request.resource.data.members.size() == 2
        && request.resource.data.members[0] is string
        && request.resource.data.members[1] is string
        && request.resource.data.members[0] < request.resource.data.members[1]
        && request.auth.uid in request.resource.data.members
        && friendshipId == canonicalChatId(request.resource.data.members[0], request.resource.data.members[1])
        && request.resource.data.createdAt is timestamp
        && acceptedRequestBetween(request.resource.data.members[0], request.resource.data.members[1]);

      // Either member can remove any checked legacy duplicate or canonical record.
      allow delete: if signedIn()
        && resource.data.members is list
        && request.auth.uid in resource.data.members;
      allow update: if false;
    }

    // One private chat per current canonical friendship.
    match /chats/{chatId} {
      allow read: if signedIn()
        && request.auth.uid in resource.data.members;

      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['members'])
        && request.resource.data.members is list
        && request.resource.data.members.size() == 2
        && request.resource.data.members[0] is string
        && request.resource.data.members[1] is string
        && request.resource.data.members[0] < request.resource.data.members[1]
        && request.auth.uid in request.resource.data.members
        && chatId == canonicalChatId(request.resource.data.members[0], request.resource.data.members[1])
        && isCurrentFriendPair(request.resource.data.members[0], request.resource.data.members[1]);

      allow update: if signedIn()
        && request.auth.uid in resource.data.members
        && isCurrentFriendPair(resource.data.members[0], resource.data.members[1])
        && request.resource.data.members is list
        && request.resource.data.members.size() == 2
        && request.resource.data.members.hasAll(resource.data.members)
        && resource.data.members.hasAll(request.resource.data.members)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members', 'lastMessage', 'updatedAt'])
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['lastMessage'])
          || (request.resource.data.lastMessage is string && request.resource.data.lastMessage.size() <= 180))
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['updatedAt'])
          || request.resource.data.updatedAt is timestamp);

      allow delete: if false;

      match /messages/{messageId} {
        allow read: if chatMember(chatId);

        allow create: if chatMember(chatId)
          && request.resource.data.keys().hasOnly(['senderId', 'senderName', 'text', 'createdAt'])
          && request.resource.data.senderId == request.auth.uid
          && request.resource.data.senderName is string
          && request.resource.data.senderName.size() <= 80
          && request.resource.data.text is string
          && request.resource.data.text.size() > 0
          && request.resource.data.text.size() <= 2000
          && request.resource.data.createdAt is timestamp;

        allow delete: if chatMember(chatId)
          && resource.data.senderId == request.auth.uid;
        allow update: if false;
      }
    }

    // Weather Pro 6.0 community observations: authenticated, short-lived, text-only reports.
    // No Firebase Storage access is needed for this feature.
    match /weatherReports/{reportId} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly([
          'uid', 'displayName', 'type', 'city', 'description',
          'lat', 'lon', 'createdAt', 'expiresAt'
        ])
        && request.resource.data.keys().hasAll([
          'uid', 'displayName', 'type', 'city', 'description',
          'lat', 'lon', 'createdAt', 'expiresAt'
        ])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() > 0
        && request.resource.data.displayName.size() <= 48
        && request.resource.data.type in ['rain', 'ice', 'fog', 'wind', 'snow', 'flood', 'other']
        && request.resource.data.city is string
        && request.resource.data.city.size() > 0
        && request.resource.data.city.size() <= 80
        && request.resource.data.description is string
        && request.resource.data.description.size() >= 3
        && request.resource.data.description.size() <= 240
        && request.resource.data.lat is number
        && request.resource.data.lat >= -90
        && request.resource.data.lat <= 90
        && request.resource.data.lon is number
        && request.resource.data.lon >= -180
        && request.resource.data.lon <= 180
        && request.resource.data.createdAt == request.time
        && request.resource.data.expiresAt is timestamp
        && request.resource.data.expiresAt > request.time
        && request.resource.data.expiresAt <= request.time + duration.value(6, 'h');
      allow delete: if signedIn() && resource.data.uid == request.auth.uid;
      allow update: if false;
    }

    // Deny every collection not explicitly granted above.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
